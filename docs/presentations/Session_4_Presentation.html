SQLite-net Setup

Database Connection and Table Creation

using SQLite;
using UnityEngine;
using System.IO;

public class SimpleDatabaseManager : MonoBehaviour
{
    private SQLiteConnection _connection;
    private string _databasePath;

    void Awake()
    {
        // For example purposes, use a simple path. In a real app, use Application.persistentDataPath
        _databasePath = "example_data.db";
        Debug.Log($"Database path: {_databasePath}");
        InitializeDatabase();
    }

    private void InitializeDatabase()
    {
        try
        {
            _connection = new SQLiteConnection(_databasePath);
            _connection.CreateTable<UserPreference>();
            _connection.CreateTable<GameScore>();
            Debug.Log("Database initialized and tables created.");
        }
        catch (SQLiteException ex)
        {
            Debug.LogError($"Failed to initialize database: {ex.Message}");
        }
    }

    void OnDestroy()
    {
        if (_connection != null)
        {
            _connection.Close();
            Debug.Log("Database connection closed.");
        }
    }
}

// Generic Example Data Models
public class UserPreference
{
    [PrimaryKey, AutoIncrement]
    public int Id { get; set; }
    public string ThemeSetting { get; set; } // e.g., "Dark", "Light"
    public float SoundVolume { get; set; } // 0.0 to 1.0
}

public class GameScore
{
    [PrimaryKey, AutoIncrement]
    public int Id { get; set; }
    public string PlayerName { get; set; }
    public int Score { get; set; }
    public string Level { get; set; } // e.g., "Forest", "Desert"
    public System.DateTime DateAchieved { get; set; }
}

CRUD Operations

Saving and Loading Player Settings

public void SavePlayerSettings(PlayerSettings settings)
{
    try
    {
        if (settings.Id == 0) // New setting
        {
            _connection.Insert(settings);
            Debug.Log($"New settings saved with ID: {settings.Id}");
        }
        else // Update existing
        {
            _connection.Update(settings);
            Debug.Log($"Settings updated for ID: {settings.Id}");
        }
    }
    catch (SQLiteException ex)
    {
        Debug.LogError($"Failed to save player settings: {ex.Message}");
    }
}

public PlayerSettings LoadPlayerSettings()
{
    try
    {
        // Assuming only one player settings entry for simplicity
        return _connection.Table().FirstOrDefault();
    }
    catch (SQLiteException ex)
    {
        Debug.LogError($"Failed to load player settings: {ex.Message}");
        return null;
    }
}

public void DeletePlayerSettings(int id)
{
    try
    {
        _connection.Delete(id);
        Debug.Log($"Player settings with ID {id} deleted.");
    }
    catch (SQLiteException ex)
    {
        Debug.LogError($"Failed to delete player settings: {ex.Message}");
    }
}

Adding and Retrieving Game Scores

public void AddGameScore(string playerName, int score, string level)
{
    var newScore = new GameScore
    {
        PlayerName = playerName,
        Score = score,
        Level = level,
        DateAchieved = System.DateTime.Now
    };
    try
    {
        _connection.Insert(newScore);
        Debug.Log($"Game score added: {playerName} - {score} on {level}");
    }
    catch (SQLiteException ex)
    {
        Debug.LogError($"Failed to add game score: {ex.Message}");
    }
}

public System.Collections.Generic.List GetTopGameScores(int count)
{
    try
    {
        return _connection.Table()
                          .OrderByDescending(s => s.Score)
                          .Take(count)
                          .ToList();
    }
    catch (SQLiteException ex)
    {
        Debug.LogError($"Failed to retrieve game scores: {ex.Message}");
        return new System.Collections.Generic.List();
    }
}
