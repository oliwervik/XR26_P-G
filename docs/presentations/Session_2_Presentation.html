Key Concepts

Singleton Pattern

Ensures a class has only one instance and provides a global point of access. Ideal for managers like a GameManager or AudioManager.

Simple Event System

Allows components to communicate without direct references, promoting loose coupling. Useful for UI updates or game state changes.

Singleton Pattern Example

Before: Direct Reference / GameObject.Find

public class Player : MonoBehaviour
{
    void Update()
    {
        // Tightly coupled and inefficient
        GameObject.Find("GameManager").GetComponent().AddScore(10);
    }
}

public class GameManager : MonoBehaviour
{
    public int score = 0;
    public void AddScore(int amount)
    {
        score += amount;
        Debug.Log("Score: " + score);
    }
}

After: Singleton Implementation

public class GameManager : MonoBehaviour
{
    public static GameManager Instance { get; private set; }
    public int score = 0;

    private void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(gameObject);
        }
        else
        {
            Instance = this;
            DontDestroyOnLoad(gameObject); // Optional: persist across scenes
        }
    }

    public void AddScore(int amount)
    {
        score += amount;
        Debug.Log("Score: " + score);
    }
}

public class Player : MonoBehaviour
{
    void Update()
    {
        if (Input.GetKeyDown(KeyCode.Space))
        {
            // Clean and globally accessible
            GameManager.Instance.AddScore(10);
        }
    }
}

Simple Event System Example

Before: Direct Method Calls

public class PlayerHealth : MonoBehaviour
{
    public int currentHealth = 100;
    public UIManager uiManager; // Direct reference

    public void TakeDamage(int amount)
    {
        currentHealth -= amount;
        uiManager.UpdateHealthDisplay(currentHealth); // Tightly coupled
        if (currentHealth <= 0)
        {
            // Handle death
        }
    }
}

public class UIManager : MonoBehaviour
{
    public Text healthText;
    public void UpdateHealthDisplay(int health)
    {
        healthText.text = "Health: " + health;
    }
}

After: UnityEvent / C# Event

using UnityEngine;
using UnityEngine.Events; // For UnityEvent

public class PlayerHealth : MonoBehaviour
{
    public int currentHealth = 100;
    public UnityEvent OnHealthChanged; // UnityEvent for inspector setup
    // public event System.Action OnHealthChanged; // Alternative: C# event

    public void TakeDamage(int amount)
    {
        currentHealth -= amount;
        OnHealthChanged?.Invoke(currentHealth); // Trigger event
        // OnHealthChanged?.Invoke(currentHealth); // For C# event
        if (currentHealth <= 0)
        {
            // Handle death
        }
    }
}

public class HealthDisplay : MonoBehaviour
{
    public Text healthText;
    public PlayerHealth playerHealth; // Reference to the event source

    void OnEnable()
    {
        playerHealth.OnHealthChanged.AddListener(UpdateHealthDisplay);
    }

    void OnDisable()
    {
        playerHealth.OnHealthChanged.RemoveListener(UpdateHealthDisplay);
    }

    void UpdateHealthDisplay(int health)
    {
        healthText.text = "Health: " + health;
    }
}

Basic Error Handling Example

Before: Unhandled Error (Null Reference)

public class PlayerController : MonoBehaviour
{
    public Rigidbody playerRigidbody; // Not assigned in Inspector

    void Start()
    {
        // This will throw a NullReferenceException if playerRigidbody is not assigned
        playerRigidbody.AddForce(Vector3.up * 10f, ForceMode.Impulse);
    }
}

After: Simple Null Check & Logging

public class PlayerController : MonoBehaviour
{
    public Rigidbody playerRigidbody;

    void Start()
    {
        if (playerRigidbody == null)
        {
            Debug.LogError("Player Rigidbody is not assigned! Please assign it in the Inspector.", this);
            return; // Stop execution if critical component is missing
        }
        playerRigidbody.AddForce(Vector3.up * 10f, ForceMode.Impulse);
    }
}
